#!/bin/sh
# ==============================================================================
# sing-box lifecycle manager for Newifi 3 D2 (MT7621AT)
# Phiên bản: ULTIMATE PATCH (AGH Bypass + TS Conflict + Lua + Full Sub)
# ==============================================================================

export PATH="/opt/sbin:/opt/bin:$PATH"

# --- Khai báo đường dẫn hệ thống ---
BIN_DIR="/tmp/sing-box"
BIN_PATH="$BIN_DIR/sing-box"
CFG_PATH="/etc/storage/singbox.conf"
LAST_MODE_FILE="/tmp/sing-box/.last_mode_memlimit"
HOT_RELOADED_MARKER="/tmp/sing-box/.hot_reloaded"
WORK_DIR="/tmp/sing-box/work"
LOG_FILE="/tmp/singbox.log"
LOG_TAG="sing-box"
WATCHDOG_FILE="/tmp/script/_opt_script_check"
ROUTE_SCRIPT="/etc_ro/singbox/singbox_route.sh"
GROUP_DIR="/tmp/sing-box/groups"
UI_DIR="/tmp/sing-box/ui"

# --- [PATCH] Danh sách Bypass hệ thống ---
SYSTEM_BYPASS_LIST='"github.com", "githubusercontent.com", "raw.githubusercontent.com", "adguard.com", "adguard-dns.com", "adguardteam.github.io", "pool.ntp.org", "time.google.com", "testingcf.jsdelivr.net", "metacubex.one"'

JQ_BIN="/usr/bin/jq"
REPO="shtorm-7/sing-box-extended"
GH_API="https://api.github.com/repos/${REPO}/releases/latest"

log() {
    logger -t "$LOG_TAG" "$1"
    echo "[$LOG_TAG] $1"
}

is_running() {
    [ -n "$(pidof sing-box)" ]
}

nv() {
    nvram get "$1" 2>/dev/null
}

# --- 1. Kiểm tra Xung đột Tailscale ---
check_ts_conflict() {
    # Nếu Tailscale Native (file tailscale_fix2.sh) đang bật hoặc đang chạy
    if [ "$(nv tailscale_enable)" = "1" ] || [ -n "$(pidof tailscaled)" ]; then
        log "CẢNH BÁO: Tailscale native đang bật -> CHẶN sing-box để tránh xung đột định tuyến!"
        # Nếu sing-box đang chạy thì phải dừng ngay
        if is_running; then
            stop purge
        fi
        return 1
    fi
    return 0
}

# --- 2. Quản lý Binary & Dashboard ---
download_binary() {
    log "Đang tải sing-box từ GitHub..."
    mkdir -p "$BIN_DIR"
    ASSET_NAME="linux-mipsle-softfloat"
    dl_url="$(curl -sk --connect-timeout 4 "$GH_API" 2>/dev/null | grep -E "browser_download_url.*${ASSET_NAME}\.tar\.gz" | cut -d'"' -f4 | head -n1)"
    [ -z "$dl_url" ] && return 1
    tmp_file="/tmp/sing-box/download_sb.tmp"
    curl -Lksfo "$tmp_file" --connect-timeout 5 "$dl_url" || wget --no-check-certificate -q -O "$tmp_file" "$dl_url"
    tar -xzf "$tmp_file" -C "$BIN_DIR/" 2>/dev/null
    extracted="$(find "$BIN_DIR" -type f -name "sing-box*" ! -name "*.tmp" | head -n1)"
    [ -n "$extracted" ] && mv -f "$extracted" "$BIN_PATH"
    chmod +x "$BIN_PATH"
    rm -f "$tmp_file"
    return 0
}

ensure_binary() {
    for p in /usr/bin/sing-box /usr/sbin/sing-box /opt/bin/sing-box; do
        if [ -x "$p" ] && "$p" version >/dev/null 2>&1; then BIN_PATH="$p"; return 0; fi
    done
    [ -f "$BIN_PATH" ] && chmod +x "$BIN_PATH" && return 0
    download_binary
}

# --- 3. Xây dựng cấu hình DNS & Route (Patch AGH) ---
build_dns_block() {
    final_detour="$1"; mode="$2"
    cat <<-EOF
  "dns": {
    "servers": [
      { "tag": "dns-remote", "type": "tls", "server": "8.8.8.8", "detour": "${final_detour}" },
      { "tag": "dns-direct", "type": "udp", "server": "1.1.1.1", "detour": "direct" },
      { "tag": "dns-fakeip", "type": "fakeip", "inet4_range": "198.18.0.0/15" }
    ],
    "rules": [
      { "domain": [${SYSTEM_BYPASS_LIST}], "server": "dns-direct" },
      { "query_type": ["A", "AAAA"], "server": "dns-fakeip" }
    ],
    "final": "dns-remote", "independent_cache": true
  },
EOF
}

build_route_block() {
    final_tag="$1"; rt_mode="$2"
    [ "$rt_mode" = "0" ] && sniff='{"inbound":"mixed-in","action":"sniff"}' || sniff='{"inbound":"tproxy-in","action":"sniff"},{"inbound":"dns-in","action":"sniff"}'
    cat <<-EOF
  "route": {
    "rules": [
      ${sniff},
      { "protocol": "dns", "action": "hijack-dns" },
      { "domain": [${SYSTEM_BYPASS_LIST}], "outbound": "direct" },
      { "ip_is_private": true, "outbound": "direct" }
    ],
    "final": "${final_tag}", "auto_detect_interface": true
  },
EOF
}

# --- 4. Quản lý Subscription (Có Lua Converter) ---
fetch_all_groups() {
    rm -rf "$GROUP_DIR"; mkdir -p "$GROUP_DIR"
    sub_list="$(nv singbox_sub_list)"
    [ -z "$sub_list" ] && return 1
    echo "[]" > "$GROUP_DIR/group_tags.json"
    
    count="$(echo "$sub_list" | "$JQ_BIN" 'length')"
    i=0
    while [ "$i" -lt "$count" ]; do
        entry="$(echo "$sub_list" | "$JQ_BIN" -c ".[$i]")"
        name="$(echo "$entry" | "$JQ_BIN" -r '.name')"; url="$(echo "$entry" | "$JQ_BIN" -r '.url')"
        i=$((i+1))
        [ "$(echo "$entry" | "$JQ_BIN" -r '.enabled')" = "false" ] && continue
        
        cache="$GROUP_DIR/sub_${i}.json"
        curl -Lksfo "$cache" --connect-timeout 5 "$url" || wget --no-check-certificate -q -O "$cache" "$url"
        
        # Lua Converter nếu không phải JSON
        if ! "$JQ_BIN" -e '.outbounds' "$cache" >/dev/null 2>&1; then
            curl -Lksfo "/tmp/sing-box/converter.lua" "https://raw.githubusercontent.com/Sophiedevops/singbox-padavan-easy-crawler-2/main/converter.lua"
            cp "$cache" /tmp/sing-box/subs_raw.txt
            cd /tmp/sing-box && lua converter.lua >/dev/null 2>&1
            [ -s "/tmp/sing-box/all_nodes.json" ] && echo '{"outbounds":' > "$cache" && cat /tmp/sing-box/all_nodes.json >> "$cache" && echo '}' >> "$cache"
        fi
        
        proxies="$("$JQ_BIN" -c --arg pfx "${name} - " '[.outbounds[]? | select(.type!="direct" and .type!="block" and .type!="dns" and .type!="selector") | .tag=($pfx+.tag)][0:100]' "$cache")"
        [ -z "$proxies" ] && continue
        echo "$proxies" | "$JQ_BIN" -c '.[]' >> "$GROUP_DIR/all_proxies.jsonl"
        group_outs="$(echo "$proxies" | "$JQ_BIN" -c '[.[].tag]')"
        "$JQ_BIN" -nc --arg tag "$name" --argjson outs "$group_outs" '{type:"urltest", tag:$tag, outbounds:$outs, url:"http://www.gstatic.com/generate_204", interval:"5m"}' >> "$GROUP_DIR/all_selectors.jsonl"
        "$JQ_BIN" -c --arg t "$name" '. + [$t]' "$GROUP_DIR/group_tags.json" > "$GROUP_DIR/group_tags.json.tmp" && mv "$GROUP_DIR/group_tags.json.tmp" "$GROUP_DIR/group_tags.json"
    done
    return 0
}

generate_config() {
    mode="$(nv singbox_mode)"
    if fetch_all_groups; then
        final_tag="select"
        master_outs="$("$JQ_BIN" -c '. + ["direct"]' "$GROUP_DIR/group_tags.json")"
        selector_block="{\\"type\\":\\"selector\\",\\"tag\\":\\"select\\",\\"outbounds\\":${master_outs}},"
        group_selectors="$(sed 's/$/,/' "$GROUP_DIR/all_selectors.jsonl")"
        outbounds_body="$(sed 's/$/,/' "$GROUP_DIR/all_proxies.jsonl")"
    else
        final_tag="direct"
    fi

    {
        echo "{"
        echo "  \"log\": { \"level\": \"info\", \"timestamp\": true },"
        echo "  \"experimental\": { \"clash_api\": { \"external_controller\": \"0.0.0.0:9090\" }, \"cache_file\": { \"enabled\": true, \"path\": \"/tmp/sing-box/cache.db\", \"store_fakeip\": true } },"
        echo "  \"inbounds\": [ {\"type\":\"tproxy\",\"tag\":\"tproxy-in\",\"listen\":\"0.0.0.0\",\"listen_port\":7893}, {\"type\":\"direct\",\"tag\":\"dns-in\",\"listen\":\"0.0.0.0\",\"listen_port\":5353} ],"
        build_dns_block "$final_tag" "$(nv singbox_dns_mode)"
        build_route_block "$final_tag" "$mode"
        echo "  \"outbounds\": [ ${selector_block} ${group_selectors} ${outbounds_body} { \"type\": \"direct\", \"tag\": \"direct\" }, { \"type\": \"block\", \"tag\": \"block\" } ]"
        echo "}"
    } | "$JQ_BIN" '.' > "$CFG_PATH"
}

# --- 5. Khởi động & Dừng ---
fix_resolv_conf() {
    if [ -f /etc/resolv.conf ]; then
        sed -i '/nameserver 127.0.0.1/d' /etc/resolv.conf
        ! grep -q "8.8.8.8" /etc/resolv.conf && echo "nameserver 8.8.8.8" >> /etc/resolv.conf
    fi
}

start() {
    # Check conflict trước khi làm bất cứ việc gì
    check_ts_conflict || return 1
    
    [ -f "$HOT_RELOADED_MARKER" ] && rm -f "$HOT_RELOADED_MARKER" && return 0
    [ "$(nv singbox_enable)" != "1" ] && return 0
    
    log "Khởi động Sing-box..."
    fix_resolv_conf
    ensure_binary
    generate_config
    
    export GOMEMLIMIT=192MiB GOGC=80
    cd "$WORK_DIR" && "$BIN_PATH" run -c "$CFG_PATH" >> "$LOG_FILE" 2>&1 &
    
    sleep 3
    if is_running; then
        [ -x "$ROUTE_SCRIPT" ] && "$ROUTE_SCRIPT" start
        # Cài Watchdog tích hợp Check Conflict
        sed -i '/sing-box/d' "$WATCHDOG_FILE" 2>/dev/null
        echo '[ -z "`pidof sing-box`" ] && /usr/bin/singbox.sh start #sing-box' >> "$WATCHDOG_FILE"
        echo '/usr/bin/singbox.sh check_ts_conflict #sing-box' >> "$WATCHDOG_FILE"
    fi
}

stop() {
    log "Dừng Sing-box..."
    sed -i '/sing-box/d' "$WATCHDOG_FILE" 2>/dev/null
    [ -x "$ROUTE_SCRIPT" ] && "$ROUTE_SCRIPT" stop
    killall sing-box 2>/dev/null
    sleep 1
    [ -n "$(pidof sing-box)" ] && killall -9 sing-box 2>/dev/null
    [ "$1" = "purge" ] || [ "$(nv singbox_enable)" != "1" ] && rm -rf "$BIN_DIR" "$LOG_FILE"
}

case "$1" in
    start) start ;;
    stop) stop "$2" ;;
    restart) stop; sleep 2; start ;;
    status) is_running && echo "running" || echo "stopped" ;;
    check_ts_conflict) check_ts_conflict ;;
    *) echo "Usage: $0 {start|stop|restart|status|check_ts_conflict}" ;;
esac
