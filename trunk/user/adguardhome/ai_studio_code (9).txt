#!/bin/sh
# ==============================================================================
# AdGuardHome Lifecycle Manager for Padavan (NEWIFI 3 D2 - MT7621)
# ==============================================================================

BIN_DIR="/tmp/AdGuardHome"
BIN_PATH="$BIN_DIR/AdGuardHome"
CFG_DIR="/etc/storage/AdGuardHome"
CFG_PATH="$CFG_DIR/AdGuardHome.yaml"
WORK_DIR="/tmp/AdGuardHome/work"
GH_API="https://api.github.com/repos/AdguardTeam/AdGuardHome/releases/latest"
GH_DL="https://github.com/AdguardTeam/AdGuardHome/releases/download"
ARCH="AdGuardHome_linux_mipsle_softfloat.tar.gz"
LOG_TAG="AdGuardHome"
WATCHDOG_FILE="/tmp/script/_opt_script_check"

log() {
    logger -t "$LOG_TAG" "$1"
}

is_running() {
    [ -n "$(pidof AdGuardHome)" ]
}

fetch_latest_tag() {
    tag="$(curl -sk --connect-timeout 5 "$GH_API" 2>/dev/null | grep '"tag_name"' | cut -d'"' -f4)"
    [ -z "$tag" ] && tag="v0.107.54"
    echo "$tag"
}

download_binary() {
    tag="$(fetch_latest_tag)"
    log "Đang tải AdGuardHome ${tag} về RAM..."
    mkdir -p "$BIN_DIR/_dl"

    url="${GH_DL}/${tag}/${ARCH}"
    archive="$BIN_DIR/_dl/agh.tar.gz"

    curl -Lksfo "$archive" --connect-timeout 10 --retry 2 --max-time 120 "$url" \
        || wget --no-check-certificate -T 20 -t 2 -q -O "$archive" "$url"

    size="$(wc -c < "$archive" 2>/dev/null)"
    if [ -z "$size" ] || [ "$size" -lt 1000000 ]; then
        log "Tải AdGuardHome thất bại!"
        rm -rf "$BIN_DIR/_dl"
        return 1
    fi

    tar -xzf "$archive" -C "$BIN_DIR/_dl" 2>/dev/null
    extracted="$(find "$BIN_DIR/_dl" -type f -name AdGuardHome | head -n1)"
    if [ -z "$extracted" ]; then
        rm -rf "$BIN_DIR/_dl"
        return 1
    fi

    mv -f "$extracted" "$BIN_PATH"
    chmod +x "$BIN_PATH"
    rm -rf "$BIN_DIR/_dl"

    log "AdGuardHome ${tag} đã sẵn sàng tại $BIN_PATH"
    return 0
}

ensure_binary() {
    if [ -x "$BIN_PATH" ] && "$BIN_PATH" --version >/dev/null 2>&1; then
        return 0
    fi
    rm -f "$BIN_PATH"
    download_binary
}

target_port() {
    case "$(nvram get adg_redirect)" in
        2) echo 53 ;;
        *) echo 5335 ;;
    esac
}

set_agh_port() {
    port="$1"
    if [ -s "$CFG_PATH" ]; then
        sed -i "/^dns:/,/^[a-zA-Z]/{s/^  port: [0-9]*/  port: ${port}/}" "$CFG_PATH"
    fi
}

dnsmasq_forward_add() {
    sed -i '/^port=0$/d; /^no-resolv$/d; /^server=127\.0\.0\.1#5335$/d' /etc/storage/dnsmasq/dnsmasq.conf
    printf 'no-resolv\nserver=127.0.0.1#5335\n' >> /etc/storage/dnsmasq/dnsmasq.conf
    /sbin/restart_dhcpd
    log "Mode 1: dnsmasq (port 53) -> chuyển tiếp sang AdGuardHome (port 5335)"
}

dnsmasq_takeover_add() {
    sed -i '/^port=0$/d; /^no-resolv$/d; /^server=127\.0\.0\.1#5335$/d' /etc/storage/dnsmasq/dnsmasq.conf
    echo 'port=0' >> /etc/storage/dnsmasq/dnsmasq.conf
    /sbin/restart_dhcpd
    log "Mode 2: dnsmasq tắt DNS, nhường trọn vẹn Port 53 cho AdGuardHome"
}

dnsmasq_restore_default() {
    sed -i '/^port=0$/d; /^no-resolv$/d; /^server=127\.0\.0\.1#5335$/d' /etc/storage/dnsmasq/dnsmasq.conf
    /sbin/restart_dhcpd
}

apply_redirect_mode() {
    mode="$(nvram get adg_redirect)"
    case "$mode" in
        1) dnsmasq_forward_add ;;
        2) dnsmasq_takeover_add ;;
        *) dnsmasq_restore_default ;;
    esac
}

wait_for_port() {
    port="$1"
    tries=0
    while [ $tries -lt 10 ]; do
        netstat -uln 2>/dev/null | grep -q ":$port " && return 0
        sleep 1
        tries=$((tries + 1))
    done
    return 1
}

wait_for_port_free() {
    port="$1"
    tries=0
    while [ $tries -lt 6 ]; do
        netstat -uln 2>/dev/null | grep -q ":$port " || return 0
        sleep 1
        tries=$((tries + 1))
    done
    return 1
}

install_watchdog() {
    [ ! -f "$WATCHDOG_FILE" ] && return
    sed -Ei "/${LOG_TAG}_watchdog/d" "$WATCHDOG_FILE"
    cat >> "$WATCHDOG_FILE" <<-EOF
[ -z "\`pidof AdGuardHome\`" ] && $0 start #${LOG_TAG}_watchdog
	EOF
}

remove_watchdog() {
    [ -f "$WATCHDOG_FILE" ] && sed -Ei "/${LOG_TAG}_watchdog/d" "$WATCHDOG_FILE"
}

generate_default_config() {
    mkdir -p "$CFG_DIR"
    cat > "$CFG_PATH" <<-EOF
http:
  address: 0.0.0.0:3030
users:
  - name: admin
    password: \$2b\$12\$zgC5mg2IyANRLOi8OHgVvePjzQ0s6uIgjlG1P3.nnzQ3ACXD9czYC
querylog:
  enabled: true
  file_enabled: true
  interval: 24h
  size_memory: 100
stats:
  enabled: true
  interval: 24h
dns:
  bind_hosts:
    - 0.0.0.0
  port: 5335
  cache_size: 1048576
  upstream_dns:
    # 127.0.0.1:5354 là cổng Inbound Real DNS của sing-box (dns-in-agh)
    - 127.0.0.1:5354
  fallback_dns:
    - https://dns.cloudflare.com/dns-query
    - 1.1.1.1
  bootstrap_dns:
    - 1.1.1.1
    - 8.8.8.8
schema_version: 28
EOF
    chmod 644 "$CFG_PATH"
    log "Đã tạo cấu hình mặc định tối ưu RAM (Dashboard cổng :3030)"
}

start() {
    if is_running; then
        return 0
    fi

    log "Đang khởi động AdGuardHome..."
    mkdir -p "$CFG_DIR" "$WORK_DIR"

    ensure_binary || return 1
    [ ! -s "$CFG_PATH" ] && generate_default_config

    mode="$(nvram get adg_redirect)"
    port="$(target_port)"
    set_agh_port "$port"

    # Nếu chạy Mode 2, giải phóng cổng 53 khỏi dnsmasq trước và đợi cổng 53 rảnh
    if [ "$mode" = "2" ]; then
        dnsmasq_takeover_add
        wait_for_port_free 53
    fi

    export GOMEMLIMIT=128MiB
    export GOGC=50

    "$BIN_PATH" -c "$CFG_PATH" -w "$WORK_DIR" --no-check-update >/dev/null 2>&1 &
    sleep 2

    if ! is_running; then
        log "LỖI: AdGuardHome không khởi động được!"
        [ "$mode" = "2" ] && dnsmasq_restore_default
        return 1
    fi
    log "AdGuardHome đã chạy (PID $(pidof AdGuardHome)), lắng nghe trên cổng $port"

    if wait_for_port "$port"; then
        [ "$mode" != "2" ] && apply_redirect_mode
    fi

    install_watchdog
    return 0
}

stop() {
    log "Đang dừng AdGuardHome..."
    remove_watchdog
    dnsmasq_restore_default

    killall AdGuardHome 2>/dev/null
    tries=0
    while is_running && [ $tries -lt 4 ]; do
        sleep 1
        tries=$((tries + 1))
    done
    is_running && killall -9 AdGuardHome 2>/dev/null

    adg_enable_now="$(nvram get adg_enable 2>/dev/null)"
    if [ "$1" = "purge" ] || [ "$adg_enable_now" != "1" ]; then
        log "Tắt hẳn AdGuardHome -> dọn sạch RAM $BIN_DIR..."
        rm -rf "$BIN_DIR"
    fi

    log "Đã dừng AdGuardHome."
}

restart() {
    stop
    sleep 1
    start
}

status() {
    is_running && echo "running" || echo "stopped"
}

case "$1" in
    start)   start ;;
    stop)    stop "$2" ;;
    restart) restart ;;
    status)  status ;;
    *) echo "Usage: $0 {start|stop|restart|status}" ;;
esac
